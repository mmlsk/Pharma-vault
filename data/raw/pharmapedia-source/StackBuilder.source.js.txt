function presetCollection(preset) {
  const text = `${preset.id} ${preset.label}`.toLowerCase();
  if (/(height|growth-plate|fgfr|achondro|gh-axis|growth-hormone|secretagogue|somatropin)/.test(text)) return "height";
  if (/(hair|skin|acne|pigment|barrier)/.test(text)) return "skin";
  if (/(focus|wake|cholinergic|all-nighter|stimulant|racetam|adaptogen|sleep)/.test(text)) return "mind";
  if (/(weight|diabetes|lipid|blood-pressure|glucose|glp1)/.test(text)) return "metabolic";
  if (/(longevity|senolytic|nad-|growth-plate|fgfr|bone|pth-|antiresorptive)/.test(text)) return "longevity";
  if (/(trt|cycle|bulk|contest|19-nor|oral-only|blast|sarm|gh-axis|pump|training|lean-mass|organ-support|cardio-repair)/.test(text)) return "performance";
  return "wellness";
}

// A preset only contributes compounds the loaded graph actually has edges for.
// interactions.json is regenerated by hand-curated batches, so a preset can
// outlive a compound id; dropping the unknown one keeps the rest of the stack
// working instead of taking the tab down on a COMPOUND_INFO lookup.
export function resolvePreset(preset, interactions) {
  return preset.compounds.filter((id) => interactions[id] && COMPOUND_INFO[id]);
}

function sameStack(a, b) {
  return a.length === b.length && a.every((id, i) => id === b[i]);
}

export function StackPresetBar({ stack, interactions, onLoad, hueOf, catColor }) {
  const stripRef = useRef(null);
  const [canGoBack, setCanGoBack] = useState(false);
  const [canGoForward, setCanGoForward] = useState(false);
  const [presetQuery, setPresetQuery] = useState("");
  const [collection, setCollection] = useState("all");
  const availablePresets = STACK_PRESETS.map((preset) => ({
    preset,
    ids: resolvePreset(preset, interactions),
    collection: presetCollection(preset),
  })).filter((entry) => entry.ids.length > 1);
  const normalizedQuery = presetQuery.trim().toLowerCase();
  const collectionPresets = collection === "all"
    ? availablePresets
    : availablePresets.filter((entry) => entry.collection === collection);
  const presets = normalizedQuery
    ? collectionPresets.filter(({ preset, ids, collection: presetCollectionId }) =>
        [preset.label, preset.blurb, preset.teaches, ...ids.map((id) => COMPOUND_INFO[id][0])]
          .join(" ")
          .toLowerCase()
          .includes(normalizedQuery) || PRESET_COLLECTIONS.find((item) => item.id === presetCollectionId)?.label.toLowerCase().includes(normalizedQuery)
      )
    : collectionPresets;
  const presetPages = Array.from({ length: Math.ceil(presets.length / 4) }, (_, index) =>
    presets.slice(index * 4, index * 4 + 4)
  );

  function updateNavigation() {
    const strip = stripRef.current;
    if (!strip) return;
    setCanGoBack(strip.scrollLeft > 2);
    setCanGoForward(strip.scrollLeft + strip.clientWidth < strip.scrollWidth - 2);
  }

  useEffect(() => {
    const strip = stripRef.current;
    if (!strip) return undefined;
    updateNavigation();
    strip.addEventListener("scroll", updateNavigation, { passive: true });
    window.addEventListener("resize", updateNavigation);
    return () => {
      strip.removeEventListener("scroll", updateNavigation);
      window.removeEventListener("resize", updateNavigation);
    };
  }, [presets.length]);

  useEffect(() => {
    const strip = stripRef.current;
    if (!strip) return;
    strip.scrollTo({ left: 0, behavior: "smooth" });
    requestAnimationFrame(updateNavigation);
  }, [normalizedQuery, collection]);

  function move(direction) {
    const strip = stripRef.current;
    if (!strip) return;
    strip.scrollBy({ left: direction * strip.clientWidth, behavior: "smooth" });
  }

  function handleStripKeyDown(event) {
    if (event.key === "ArrowLeft" || event.key === "ArrowRight") {
      event.preventDefault();
      move(event.key === "ArrowLeft" ? -1 : 1);
    } else if (event.key === "Home" || event.key === "End") {
      event.preventDefault();
      stripRef.current?.scrollTo({
        left: event.key === "Home" ? 0 : stripRef.current.scrollWidth,
        behavior: "smooth",
      });
    }
  }

  if (availablePresets.length === 0) return null;

  return html`
    <div class="stack-presets">
      <div class="stack-presets__head">
        <div class="stack-presets__heading">
          <span class="stack-presets__label">Start from a common stack</span>
          <span class="stack-presets__note">
            Reference combinations, not recommendations — no doses, and some are here because they score badly.
          </span>
        </div>
        <div class="stack-presets__nav" aria-label="Navigate common stacks">
          <label class="stack-presets__search">
            <input
              type="search"
              value=${presetQuery}
              onInput=${(event) => setPresetQuery(event.target.value)}
              placeholder="Find a stack…"
              aria-label="Search common stacks"
            />
          </label>
          <span class="stack-presets__result-count" aria-live="polite">${presets.length}/${availablePresets.length}</span>
          <button
            type="button"
            class="stack-presets__arrow"
            aria-label="Previous stacks"
            title="Previous stacks"
            disabled=${!canGoBack}
            onClick=${() => move(-1)}
          >←</button>
          <button
            type="button"
            class="stack-presets__arrow"
            aria-label="Next stacks"
            title="Next stacks"
            disabled=${!canGoForward}
            onClick=${() => move(1)}
          >→</button>
        </div>
      </div>
      <div class="stack-presets__collections" aria-label="Filter preset collections">
        ${PRESET_COLLECTIONS.map((item) => {
          const count = item.id === "all" ? availablePresets.length : availablePresets.filter((entry) => entry.collection === item.id).length;
          return html`<button
            key=${item.id}
            type="button"
            class=${collection === item.id ? "is-active" : ""}
            aria-pressed=${collection === item.id}
            onClick=${() => setCollection(item.id)}
          ><span>${item.label}</span><small>${count}</small></button>`;
        })}
      </div>
      <div
        class=${`stack-presets__strip ${presets.length === 0 ? "stack-presets__strip--empty" : ""}`}
        ref=${stripRef}
        tabindex="0"
        aria-label="Common stack presets. Use left and right arrow keys to navigate."
        onKeyDown=${handleStripKeyDown}
      >
        ${presets.length === 0
          ? html`<div class="stack-presets__no-results">
              <b>No matching stacks</b>
              <span>Try a compound name, goal, or mechanism.</span>
              <button type="button" onClick=${() => setPresetQuery("")}>Clear search</button>
            </div>`
          : presetPages.map((page, pageIndex) => html`
              <div class="stack-presets__page" key=${pageIndex}>
                ${page.map(({ preset, ids, collection: presetCollectionId }) => {
                  const active = sameStack(stack, ids);
                  const collectionLabel = PRESET_COLLECTIONS.find((item) => item.id === presetCollectionId)?.label;
                  return html`
                    <button
                      key=${preset.id}
                      class=${`stack-preset ${active ? "stack-preset--active" : ""}`}
                      onClick=${() => onLoad(ids)}
                      title=${preset.teaches}
                    >
                      <span class="stack-preset__top">
                        <span><small class=${`stack-preset__kind stack-preset__kind--${presetCollectionId}`}>${collectionLabel}</small><span class="stack-preset__label">${preset.label}</span></span>
                        <span class="stack-preset__count">${ids.length}</span>
                      </span>
                      <span class="stack-preset__blurb">${preset.blurb}</span>
                      <span class="stack-preset__compounds">
                        ${ids.map((id) => COMPOUND_INFO[id][0].split(" (")[0]).join(" · ")}
                      </span>
                      <span class="stack-preset__lesson"><b>Shows:</b> ${preset.teaches}</span>
                      <span class="stack-preset__dots">
                        ${ids.map(
                          (id) => html`<span
                            key=${id}
                            class="stack-preset__dot"
                            style=${{ background: catColor(hueOf(id)) }}
                            title=${COMPOUND_INFO[id][0]}
                          ></span>`
                        )}
                      </span>
                      <span class="stack-preset__load">${active ? "Loaded" : "Load stack"}<span aria-hidden="true">${active ? "✓" : "→"}</span></span>
                    </button>
                  `;
                })}
              </div>
            `)}
      </div>
    </div>
  `;
}

export const SHOP_GROUP_A_SHIPPING_CENTS = 5_000;

export const SHOP_GROUP_D_SHIPPING_CENTS = 4_000;

export function calculateShopChargesCents(subtotalCents, shippingGroups = []) {
  const groups = Array.isArray(shippingGroups)
    ? shippingGroups
    : String(shippingGroups || "").split("+");
  const shippingCents = (groups.includes("A") ? SHOP_GROUP_A_SHIPPING_CENTS : 0)
    + (groups.includes("B") ? SHOP_GROUP_B_SHIPPING_CENTS : 0)
    + (groups.includes("C") ? SHOP_GROUP_C_SHIPPING_CENTS : 0)
    + (groups.includes("D") ? SHOP_GROUP_D_SHIPPING_CENTS : 0)
    + (groups.includes("E") ? SHOP_GROUP_E_SHIPPING_CENTS : 0)
    + (groups.includes("G") ? SHOP_GROUP_G_SHIPPING_CENTS : 0);
  const carrierFeeCents = Math.round((subtotalCents + shippingCents) * SHOP_CARRIER_FEE_PERCENT / 100);
  return {
    subtotalCents,
    shippingCents,
    carrierFeeCents,
    totalCents: subtotalCents + shippingCents + carrierFeeCents,
  };
}
